-- Find Top Behavioral Twins across 100 Candidates for Active Benchmark
WITH ActiveBenchmark AS (
  SELECT features 
  FROM `population_dynamics___us.v1_s2_z12` 
  WHERE geo_id = '80ead45'
)
SELECT 
  candidate.candidate_id,
  candidate.city,
  candidate.state,
  candidate.geo_id,
  -- Cosine similarity: (1 - Cosine Distance) * 100
  ROUND((1 - ML.DISTANCE(ActiveBenchmark.features, candidate.features, 'COSINE')) * 100, 1) AS similarity_match_pct
FROM `qsr_expansion.candidates_100` candidate, ActiveBenchmark
ORDER BY similarity_match_pct DESC
LIMIT 10;
